\begin{frame}{Cleaning, features and leakage}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \textbf{Cleaning (every step logged)}
    \begin{itemize}\small
      \item 33k re-scrapes of the same URL removed (newest kept), plus 709 re-posts
      \item Strings $\to$ numbers: ``145 m²'', ``7 / 9'', ``1.3 sot''
      \item Fixed domain outlier rules (no quantiles $\Rightarrow$ nothing learned from test)
      \item Imputation medians and one-hot vocabularies from \emph{train only}
    \end{itemize}
    \column{0.5\textwidth}
    \textbf{Removed: leakage and identifiers}
    \begin{itemize}\small
      \item \alert{\texttt{unit\_price}} = price / area
      \item \alert{\texttt{total\_price}} = price
      \item owner/shop names, address, ids, URLs
      \item description (restates the price), \texttt{views} (arrives after listing)
    \end{itemize}
    \textbf{\NFeat{} features:} area, rooms, floors, land, lat/lng, distance to centre,
    amenities, one-hot category / district / neighbourhood
  \end{columns}

\begin{frame}{Our decision tree}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \begin{itemize}\small
      \item Gini, entropy (classification); MSE (regression)
      \item $\mathrm{IG}=I(\text{parent})-\sum_c \frac{N_c}{N}I(c)$
      \item Per feature: sort once, prefix sums $\Rightarrow$ score \emph{all} thresholds in one vectorised pass
      \item Stopping: \texttt{max\_depth}, \texttt{min\_samples\_split}, \texttt{min\_samples\_leaf}, \texttt{min\_impurity\_decrease}
      \item Tests: \textbf{identical splits to sklearn} on a toy example; every hyperparameter changes the tree
      \item Bug found on real data: the midpoint of adjacent floats rounds up $\Rightarrow$ an empty child
    \end{itemize}
    \column{0.52\textwidth}

\begin{frame}{Our Pegasos SVM}
  \begin{columns}[T]
    \column{0.45\textwidth}
    \small
    $J=\frac{\lambda}{2}\|\mathbf w\|^2+\frac1n\sum_i\max(0,1-y_i\mathbf w^\top\mathbf x_i)$\\[4pt]
    $\mathbf w\leftarrow(1-\eta_t\lambda)\mathbf w+\frac{\eta_t}{|B|}\sum_{i\in A}y_i\mathbf x_i,\ \eta_t=\frac1{\lambda t}$
    \begin{itemize}\small
      \item Mini-batch, projection, iterate averaging
      \item Margin $2/\|\mathbf w\|$, support vectors $y_if(\mathbf x_i)\le1$
      \item Non-linear: \textbf{random Fourier features} ($O(nD)$ vs $O(n^2)$ for kernel Pegasos)
      \item Chosen $\lambda=\SvmLam$: $\|\mathbf w\|=\SvmNormW$, margin \SvmMargin, \SvmSvFrac\% SVs
    \end{itemize}
    \column{0.55\textwidth}
    \includegraphics[width=\linewidth]{svm_margin_2d.png}
    \includegraphics[width=\linewidth]{svm_lambda_sweep.png}
  \end{columns}
\end{frame}

\begin{frame}{Experimental setup}
  \begin{itemize}
    \item Tuning on \textbf{validation only}; the test set is used once
    \item Tree A: depth \TreeADepth, leaf $\ge$ \TreeALeaf \quad Tree B: \TreeBCrit, depth \TreeBDepth, leaf $\ge$ \TreeBLeaf
    \item SVM: $\lambda=\SvmLam$ \quad RFF: $\gamma=\RffGamma$, $\lambda=\RffLam$, $D=1000$
      (most regularised within 0.002 F1 of the best)
    \item Baselines with \textbf{matched} settings: sklearn trees; \texttt{Pipeline(StandardScaler, SGDClassifier(hinge, alpha=}$\lambda$\texttt{))};
      SVC linear and RBF with $C=1/(\lambda n)$ (10k subsample)
    \item Uncertainty: bootstrap std on test, plus 5-fold CV mean $\pm$ std on train+val
  \end{itemize}
  \centering

\end{frame}

\begin{frame}{Ours vs scikit-learn: why the small gaps?}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \resizebox{\linewidth}{!}{\input{../report/tables/tree_structure.tex}}
    \begin{itemize}\small
      \item Same root split, 6--7 of the first 7 splits equal; \SameA\% / \SameB\% identical test predictions
      \item Gaps: sklearn uses float32 thresholds; random feature order on ties; divergence after a deep disagreement
      \item Speed: our tree \OursTreeAFit\,s vs \SkTreeAFit\,s (Cython); our Pegasos \OursSvmFit\,s

\begin{frame}{Where the models fail}
  \begin{columns}[T]
    \column{0.55\textwidth}
    \includegraphics[width=\linewidth]{errors_regression.png}
    \includegraphics[width=0.85\linewidth]{errors_classification_threshold.png}
    \column{0.45\textwidth}
    \begin{itemize}\small
      \item \textbf{Tails:} leaf means shrink to the middle, so luxury is under-predicted and bargains over-predicted
      \item \textbf{Houses} are much harder than flats; district error tracks house share ($r=\DistHouseR$) more than district size ($r=\DistSizeR$)
      \item \textbf{Tier:} \SvmErrNear\% of SVM errors are within $\sim$20\% of the threshold, and many are irreducible
      \item Limits: asking prices only; one 5-week snapshot; random (not grouped) split; text unused
    \end{itemize}
  \end{columns}
